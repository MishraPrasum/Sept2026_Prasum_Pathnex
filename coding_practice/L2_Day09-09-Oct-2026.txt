# Oct L2_Day09 - Docker integration with Jenkins and GitLab 

- name: Install docker container for nginx
  hosts: all
  become: yes
 
  tasks:
	- name: Pull Nginx image
	  docker_image:
	     name: nginx	 
	     source: pull
	- name: Run Nginx Container
	  docker_container:
	      name: nginx-container
	      image: image
	      state: started
	      published_ports:
	   	  - "8080:80"

# Kubernetes - Horizontal Pod Autoscaling (HPA)

apiversion: apps/v1
kind: deployment
metadata:
	name: pathnex-deployment
spec:
  replicas: 1
  selector:
    matchlabels:
	app: pathnex-app	
  template:
    metadata:
	labels:
	  app: pathnex-app
     spec:
	containers:
	    name: app-nginx
	    image: nginx
	    ports:
	      - containerPort: 80

---
apiversion: autoscaling/v2
kind: Horizontal Pod Autoscalaer
metadata:
	name: pathnex-hpa
spec:
    scaleTargetRef:
	apiversion: apps/v1
  	kind: deployment
	name: pathnex-deployment
    minreplicas: 1
    maxreplicas: 10
    targetCPUUtilizationPercentage: 50



# Jenkins pipeline - Deploy to kubernetes

pipeline {
	agent any 
	stages {
	   stage('Deployment to kubernetes') {
		steps {
		  script {
		   sh 'kubectl apply -f deployment.yaml'
		      }
		}
	   }
	}
  }


GitLab CI/CD - Deploy to kubernetes
