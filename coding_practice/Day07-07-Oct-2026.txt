# Kubernetes - Deployment with 2 replicas

apiversion: apps/v1
kind: Deployment
metadata:
	name: pathnex-app
spec:
   replicas: 2
   selector:
      matchlabels:
	 app: pathnex-app
   template:
      metadata:
	 labels:
	    app: pathnex-app
      spec:
	 containers:
	    - name: app
	      image: nginx
	      ports:
	        - containerPort: 80

# Jenkins pipeline - Parameterized

pipeline {
	agent any
	parameters {
	   string(name: 'ENVRIONMENT', defaultValue: 'dev', description: 'Deployment Environment')
	}
	environment {
	  INSTITUTE_NAME = "pathnex"
	}
	stages {
	   stage('Checkout') {
		steps {
		   git url: 'https://github.com/Pathnex/sample-java-app.git'
		 }
	   }
	   stage('Deploy') {
		steps{
		   sh 'echo "Deployment to $ENVIRONMENT by $INSTITUE_NAME"'
		}
	   }
      }
}
