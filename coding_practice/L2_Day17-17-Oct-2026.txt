# Ansible — Auto-healing Web Servers with Systemd

- name: Configure Auto-healing for web server
  hosts: all
  become: yes

  tasks:
	- name: Ensure nginx service is running
	  service:
	      name: nginx
	      state: started
	      enabled: yes
	- name: Setup auto-restart for nginx service
	  lineinfile:
	      path: /etc/systemd/system/nginx.service.d/override.conf
	      line: "Restart=always"
	      create: yes
	- name: Reload systemd
	  command: systemctl daemon-reload

# Kubernetes - deployment with liveness and rediness 

apiversion: apps/v1
kind: Deployment
metadata:
	name: pathnex-web-app
spec:
  replicas: 3
  selector:
     matchlabels:
	app: pathnex-web-app
  template
   metadata: 
     labels:
	app: pathnex-web-app
    spec:
	containers:
	   - name: web
	     image: nginx:latest
	     ports:
	       - containerPort: 80
	     livenessProbe:
		httpGet:
		   path: /healthz
		   port: 80
		initalDelaySeconds: 5
		periodSeconds: 10
	     redinessProbe:
		httpGet:
		   path: /healthz
		   port: 80
		initalDelaySeconds: 5
		periodSeconds: 10 

# JenkinsFile - Rolling Deployments with Kubernetes

pipeline {
	agent any
	environment {
	   IMAGE_NAME = "pathnex-web-app"
	}
	stages{
	  stage('Build') {
	    steps {
		sh 'docker.build('pathnex-web-app')'
		}
	  }
	  stage('Test') {
	    steps {
		echo "Running Test"
		}
	  }
	  stage('Deploy') {
	     steps {
		script {
		   sh 'kubectl apply -f kubernetes/deployment.yaml'
		   sh 'kubectl rollout status deployment/pathnex-web-app'
			}			
		}
	}
    }	
}



# GitLab CI/CD — Multi-Cloud Deployment

stages:
   - build
   - push
   - deploy

build:
	stage: build
	script:
	   - docker build -t pathnex-web-app .

push:
	stage: push
	script:
	   - docker login -u "$CI_REGISTRY_USER" -u "$CI_REGISTRY_PASSWORD"
	   - docker push pathnex-web-app
deploy_aws:
	stage: deploy
	script:
	  - kubectl --context=aws-cluster apply -f kubernetes/deployment.yaml
deploy_azure:
	stage: deploy
	script:
	  - kubectl --context=azure-cluster apply -f kubernetes/deployment.yaml
