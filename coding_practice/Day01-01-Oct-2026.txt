# Ansible Task - Install Nginx on Pathnex Server

- name: Install nginx on pathnex server
  hosts: all
  become: yes

  tasks:
	- name: Install Nginx
	  yum:
 	    name: nginx
	    state: present

# Kubernetes Task - Create Nginx Pod

apiversion: v1
kind: Pod
metadata:
	name: nginx-pod
spec:
  containers:
     - name: nginx
       image: nginx:latest
       ports:
	 - containerPort: 80

# Jenkins pipeline - Git Integration 

pipeline {
	agent any
	stages {
	  stage('Checkout') {
	    steps {
		git branch: 'main', url: 'https://github.com/Pathnex/sample-java-app.git'
		}
	 }
	  stage(List files) {
	   steps {
	    sh 'ls -la'
	       }
	}	
   }

}


# GitLab CI - Git Checkout

stages:
  - checkout

git-checkout:
    stage: checkout
    script:
	- git clone https://github.com/Pathnex/sample-java-app.git
	- cd sample-java-app
	- ls -la
