# Oct L2_Day07 - Automating cloud infrastructure with Terraform & Ansible

# Ansible - Setup nginx with custom config

- name: setup nginx with custom config
  hosts: all
  become: yes
  
  tasks:
	- name: install nginx
	  yum:
	     name: nginx
	     state: present
	- name: copy custom nginx.conf
	  copy:
	    src: nginx.conf
	    dest: /etc/nginx/nginx.conf
	- name: start nginx service
	  service:
	     name: nginx
	     state: started
	     enabled: yes


# Terraform - Provision EC2 with IAM role

resource "aws_iam_role" "ec2_role" {
	name = "pathnex_ec2_role"
	assume_role_policy = jsonencode({
	   version = "2012-10-17"
	   statement = [
	      {
		Action = "sts:AssumeRole"
	        Effect = "Allow"
		principal = {
		   service = "ec2.amazonaws.com"
		}
	},
    ]

})

}


resource "aws_instance" "pathnexEc2" {
	ami = "ami-0abcd1234abcd1234"
	instance_type = "t2.medium"
	iam_instance_policy = aws_iam_role.ec2_role.name
	tags = {
	  Name = "Pathnex-EC2"
 	}
}
