# Ansible:

- name: Install Java and set Java_Home
  hosts: web_server
  become: yes

  tasks:
	- name: Install Java
	  yum:
 	     name: java-11-openjdk
	     state: present
	- name: Set Java_Home
	  lineinfile:
	      path: /etc/profile
	      line: 'export JAVA_HOME=/usr/lib/jvm/java-11-openjdk'

#Jenkins Pipeline — Slack Notification

pipeline {
	agent any
	environment {
	   INTITUTE_NAME = "Pathnex"
	   SLACK_CHANNEL = "#devops"	
	}
	stages {
	   stage('Checkout') {
	     steps {
		git branch: 'main', url: 'https://github.com/mishraprasum/terragrunt.git'		
		}
	  }
	  stage('Build') {
	     steps {
		steps {
		   sh 'mvn clean package'
			}
		}
	 }
	 post {
		success {
		    slackSend(channel: "#SLACL_CHANNEL", message: "Success: $INSTITUTE_NAME Build #${env.build_number}")
		}
	 	failure {
		    slackSend(channel: "#SLACK_CHANNEL", message: "Failure: $INSTITUTE_NAME Build #${env.build_number}")
		}
	  }
     }

}
