  tasks:
	- name: launch EC2 instance in us-west-1

	      name: PathnexEC2-us-west-1

	      region: us-west-1
	      instance_type: t3.micro
	      image: ami-0abcd1234abcd1234

	  ec2_instance:
		name: pathnexEC2-us-east-1
		key_name: pathnex-key
		region: us-east-1
		image: ami-0abcd1234abcd1234
		wait: yes



# Terraform - Multi-Region VPC with EC2

resource "aws_vpc" "us_east_vpc" {
	cidr_bloc = "10.0.0.0/16"
	region = "us-east-1"
}

resource "aws_vpc" "use-west_vpc" {
	cidr_block = "10.1.0.0/16"
	region = "use-west-1"
}


resource "aws_instance" "ec2_us_east" {
	ami = "ami-0abcd1234abcd1234"
	instance_type = "t3.micro"
	region = "us-east-1"
}

resource "aws_instance" "ec2_us_west" {
	ami = "ami-0abcd1234abcd1234"

}


#Kubernetes — Multi-Cluster Deployment with Federation

apiversion: apps/v1
kind: Deployment
metadata:
	name: pathnex-federation
spec:
  replicas: 2
  selector:
    matchlabels:
	app: pathnex-app
  template:
    metadata:
	labels:
	   app: pathnex-app
    containers:
	- name: nginx
	  image: nginx
	  ports:
	     - containerPort: 80

---
apiversion: v1
kind: Service
metadata:
	name: pathnex-service
spec:
   type: ClusterIP
   selector:
	app: pathnex-app
   ports:
	- protocol: TCP
	  port: 80


# Jenkinsfile — Deploy to Multi-Cluster Environments

pipeline {
	agent any
	stages{
	  stage('Build') {
	     steps {
		echo "Building docker image.."
		docker.build('pathnex-web-app')
		}
   	  }
	  stage('Test') {
		steps{
		   echo "Running Tests"
		   }
	  }
	  stage('Deploy to Kubernetes') {
		steps {
		  script {
			if (env.cluster == 'us-east-1') {
			   sh 'kubectl --context=us-east apply -f kubernetes/deployment.yaml'
			} else {
			   sh 'kubectl --context=us-west apply -f kubernetes/deployment.yaml'
				}
			}
		} 	
	 }
     }
}


# GitLab CI/CD — Multi-Region Deployment with Kubernetes

stages:
   - build
   - push
   - deploy


build:
	stage: build
	script:
	  - docker build -t pathnex-web-app .
	
push:
	stage: push
	script:
	  - docket login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
	  - docker push pathnex-web-app
deploy:
	stage: deploy
	script:
	   - |
	     if [ "$CI_COMMIT_REF_NAME" == "main" ] then
		kubectl --context=us-east apply -f kubernetes/pro-deployment.yaml
	     else
		kubectl --context=us-west apply -f kubernetes/dev-deployment.yaml
	     fi
