# L2_Day03 - Ansible, Terraform, Kubernetes basics

# Ansible - Install and Start Apache

- name: Install and start Apache
  host: all
  become: yes

  tasks:
	- name: Install apache
  	  yum:
	    name: httpd
	    state: present
	- name: Start apache
	  service:
	     name:httpd
	     state: started
	     enabled: yes
