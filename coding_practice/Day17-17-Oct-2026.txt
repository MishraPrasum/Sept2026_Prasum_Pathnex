  become: yes

	users:
	   - dev1
	   - dev2
	   - dev3

  tasks:
	- name: Create users

	     name: "{{ item }}"
	     state: present
	  loop: "{{ users }}"

# Jenkins Pipeline — Simulated Deployment

	agent any
	parameters{
	   choice(name: 'ENVRIONMENT', choices: ['dev', 'staging', 'prod'], description: 'Deployment Environment')
	}
	environment {
	   INSTITUTE_NAME = "Pathnex"
	}
	stages {
	   stage('Deploy') {
		steps {
		   sh 'echo "Deploying $INSTITUTE_NAME application to $ENVIRONMENT"'
		     }
	   }
	}	
}
