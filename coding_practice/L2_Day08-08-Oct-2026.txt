# Ansible - Install Docker compose

- name: Install Docker compose
  hosts: all
  become: yes
  
  tasks:
	- name: install dependencies
	  yum:
	     name: "python3-pip"
	     state: present
	- name: install docker-compose
	  pip:
	    name: docker-compose


# Terraform - EC2 with Security group and EBS volume

resource "aws_security_group" "allow_ssh_http" {
	name = "allow_ssh_http"
	descriptionn  = "Allow SSH and HTTP access"

	ingress {
	  from_port = 22
	  to_port   = 22
	  protocol  = "tcp"
	  cidr_block = ["0.0.0.0/0"]
	}

	ingress {
	  from_port = 80
	  to_port   = 80
	  protocol  = "tcp"
	  cidr_block = ["0.0.0.0/0"]
	}
}

resource "aws_instance" "pathnexEC2" {
	ami  = "ami-0abcd1234abcd1234"
	instance_type  = "t3.micro"
	security_group  = [aws_security_group.allow_ssh_http.name]
	
	ebs_block_device {
	   device_name = "/dev/sdh"
	   volume_size = 30
	}
}

# Jenkins Pipeline - Docker build and Push

pipeline {
	agent any 
	stages {
	   stage('Build Docker Image') {
	      steps {
		script {
		   docker.build('pathnex-app')
			}
		}
	   }
	   stage('Push Docker image') {
		steps {
		   script {
			docker.withRegistry('https://docker.io', 'docker-credentials') {
			docker.image('pathnex-app').push('latest')
			 }
		    }
		}
           }
     }
}



# Gitlab CI/CD - Docker Build and Push

stages:
   - build
   - push

build:
	stage: build
	script:
	   - docker build -t pathnex-app .

push:
	stage: push
	script:
	   - docker login -u $CI_REGISTRY_USER -p $CI_REGISTRY_PASSWORD
	   - docker push pathnex-app
