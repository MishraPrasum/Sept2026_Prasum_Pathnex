# Jenkins pipeline - with git integration and testing

pipeline {
	agent any
	stages {
	  stage('Clone Repository') {
	     steps {
		git 'https://github.com/pathnex/sample-repo.git'
		   }
	   }
	  stage('Build') {
	     steps {
		echo "Building project from git repository"
		}
	  }
	  stage('Test') {
	     steps {
		echo "Running Unit test"
		}
	    }
	  stage('Deploy') {
		steps {
		  echo "Deploying application"
		}
	    }
	}
}


# GitLab CI/CD - with Docker build

   - build
   - push
   - deploy

   stage: build
   script:
      - docker build -t pathnex-app .

push:
   stage: push
   script:
	- docker push pathnex-app

deploy:
   stage: deploy
   script:
      - kubctl apply -f kubernetes/deployment.yaml



# Ansible - Create user and set permissions

- name: Create user and set permission
  hosts: all
  become: yes

  tasks:
	- name: create user
  	  user:
	     name: pathnex
	     state: present
	     shell: /bin/bash
	- name: set permissions for the user
	  file:
	    path: /home/pathnex
	    owner: pathnex
	    group: pathnex
	    mode: '0755'


# Terraform - EC2 with tags and Volume
