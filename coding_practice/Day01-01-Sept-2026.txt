## Ansible Task — Install Nginx on Pathnex Server

- name: Install Nginx on Pathnex Server
  hosts: all
  become: yes

  tasks:
	- name: Install Nginx
	  yum:
	    name: nginx
	    state: present

## Kubernetes Task - Create Nginx Pod

apiversion: v1
kind: deployment
metadata:
	name: pathnex-nginx
	namespace: global
spec:
  replicas: 2
  selector:
     matchlabels:
	app:nginx

  template:
     metadata:
	labels:
	  app:nginx
	  name: nginx-deploy-pod
spec:
   containers:
	- name: Web-Nginx
	  image: nginx:latest
	  ports:
	    - containerport: 80
