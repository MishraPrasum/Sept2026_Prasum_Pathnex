# Ansible:
- Install nginx
- Deploy "Welcome to Pathnex" HTML page
- Start & enable service

---
- name: Install Nginx on Pathnex Server
  hosts: all
  become: yes
  
  tasks:
	- name: Install Nginx
	  yum:
	    name: nginx
   	    state: present
	- name: Deploy HTML Page
	  copy:
	  content: |
	  <!DOCTYPE html>
	  <html>
	  <body>
	  <h1> Welcome to Pathnex</h1>
	  </body>
	  </html>
	  dest: /usr/nginx/html/index.html
	  owner: root
	  group: root
	  mode: '0644'
	- name: Start Nginx service
	  service:
	     name: nginx
	     state: started
	     enabled: yes




# Kubernetes:
- Create Deployment with PVC
- Expose via NodePort

# Create the PVC

apiversion: v1
kind: PersistentVolumeClaim
metadata:
   name: pathnex-pvc
spec:
  accessModes:
     - ReadWriteOnce
  resources:
    requests:
      storage: 1Gi

---
apiversion: apps/v1
kind: Deployment
metadata:
	name: pathnex-nginx
spec:
  replicas: 3
  selector:
    matchlabels:
	app: pathnex-nginx
  template:
    metadata:
	labels:
	  app: pathnex-nginx
   spec:
      containers:
	 - name: nginx 
	   image: nginx:latest
	   ports:
	      - containerport: 80
           volumeMounts:
	      - name: nginx-storage
	        mountPath: /usr/share/nginx/html
      volumes:
	- name: nginx-storage
	  persistentVolumeClaim
	     claimName: pathnex-pvc

---
apiversion: v1
kind: service
metadata:
	name: pathnex-nodeport
spec:
  -Type: Nodeport
   selector:
      app: pathnex-nginx
   ports:
      port: 80
      targetport: 80

# Jenkins Pipeline:
You will learn to **combine all tasks into a realistic Jenkins Pipeline** with proper stages and tags.

pipeline {
agent any

  environment {
    INSTITUTE_NAME = "Pathnex"
    TEAM           = "DevOps"
    ENV            = "prod"
    PROJECT        = "Pathnex-Training"
  }

   stages {
    stage('Infrastructure') {
        steps {
            echo "Creating Infrastructure..."

            sh '''
                terraform init
                terraform validate
                terraform plan
                terraform apply -auto-approve
            '''
        }
    }

    stage('App Deployment') {
        steps {
            echo "Deploying Application..."

            sh '''
                ansible-playbook -i inventory ansible/nginx.yml

                kubectl apply -f kubernetes/pvc.yaml
                kubectl apply -f kubernetes/deployment.yaml
                kubectl apply -f kubernetes/service.yaml
            '''
        }
    }

    stage('Monitoring Checks') {
        steps {
            echo "Running Monitoring Checks..."

            sh '''
                echo "$(top)"

                echo "$(free -h)

                echo "$(df -h)
                
                echo "$(hostname -I)

                echo "$(ps | head -n 5)
            '''
 	       }
    	}
    }

    post {
    success {
        echo "Deployment Completed Successfully! of $INSTITUTE_NAME on $ENV for $TEAM and $PROJECT"
    }

    failure {
        echo "Pathnex Deployment Failed! of $INSTITUTE_NAME on $ENV for $TEAM and $PROJECT"
    }
  }

}

}
