# Ansible - Install Helm on server
 
- name: Install Helm
  host: all
  become: yes

  tasks:
	- name: Download Helm 
	  get_url:
	      url: https://get.helm.sh/helm-v3.7.0-linux-amd64.tar.gz
	      dest: /temp/heml.tar.gz
	- name: Extract helm
	  unarchive:
	      src: /temp/heml.tar.gz
	      dest: /user/local/bin
	      remote_src: yes
	- name: binary permission helm file
	  file:
	      path: /user/local/bin/helm
	      mode: '0755'

Jenkins - multi-stage pipeline

pipeline {
	agent any
	environment {
	   INSTITUTE_NAME = "Pathnex-web-app"
	}
	stages {
	   stage('Checkout') {
		steps {
		   git url: 'https://github.com/pathnex/sample-repo.git'
		    }
	   }
	   stage('Build Docker Image') {
		steps {
		   script {
			docker.build("${env.IMAGE_NAME}")
			}
		    }
	    }
	   stage('Push Docket image') {
		steps {
		  script {
			docker.withRegistry('http://docker.io', 'docker_credentials') {
			docker.image("${env.IMAGE_NAME}").push('latest')
				}
		         }
		}
	   }
	   stage('Deploy to Kubernetes') {
		steps {
		   script {
			kubectl apply -f deployment.yaml
			}
		 }
	  }
     }

}



GitLab CI/CD - Multi Stage pipeline


stages:
   - build
   - push
   - deploy

build:
	stage: build
	script:
	   - docker build -t pathnex-web-app .

push:
	stage: push
	script:
	   - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
	   - docker push pathnex-web-app

deploy:
	stage: deploy
	script:
	  - helm repo add stable https://charts.helm.sh/stable
	  - helm install pathnex-nginx stable/nginx-ingress
