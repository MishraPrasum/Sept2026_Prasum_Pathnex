# Ansible - Deploy a web application to multiple host
 
- name:Deploy a web application to multiple host
  hosts: web_server
  become: yes
  tasks:
	- name: Deploy a web app
	  copy:
	    src: /path/to/web/app/
	    dest: /var/www/html/
	- name: Start web app service
	  service:
	    name: appache2
	    state: started
	    enabled: yes

# Jenkinsfile — Multi-Environment Deployment

pipeline {
	agent any
	environment {
	  IMAGE_NAME = "pathnex-web-app"
	  DEPLOY_ENV = "production"
	}
	stages {
	  stage('Build') {
		steps {
		   echo "building docker image"
		   docker.build("${env.IMAGE_NAME}")		
		}
	}
	  stage('Test') {
		steps {
		   echo "Running Tests"
 		}
	 }
	  stage('Deploy to k8s') {
		steps {
		   script {
		       if (env.DEPLOY_ENV == 'Production') {
			  sh 'kubectl apply -f kubernetes/prod-deployment.yaml'
			}
		       else {
			 sh 'kubctl apply -f kubernetes/dev-deployment.yaml'
			}		
		    }
		}
          }
     }
}
