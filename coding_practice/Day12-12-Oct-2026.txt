# kubernetes - Rolling update strategy

apiversion: apps/v1
kind: Deployment
metadata:
	name: pathnex-rolling
	namespace: pathnex
spec:
   replicas: 3
   strategy:	
	type: RollingUpdate
	rollingupdate:
	   maxSurge: 1
	   maxunavailable: 1
   selector:
     matchlabels:
	app: pathnex-web
   template:
     metadata:
	labels:
	   app: pathnex-web

     spec:
	containers:
	   - name: web
	     image: nginx



# Shell script - backup directory

#!/bin/bash

SOURCE="/var/log"
DEST="/backup/prasum-$(date +%F).tar.gz"

tar -czf $DEST $SOURCE
echo "backup created at: $DEST"



# Jenkins Pipeline — Multi-branch

pipeline {
	agent any
	environment {
	  INSTITUTE_NAME = "Pathnex"
	}
	stages{
	  stage('Checkout') {
	     steps {
	        checkout scm  # Checks out the source code from repo configured for the pipeline
		}
	}
	 stage('Build') {
	    steps {
		sh  'mvn clean compile'
		}
	}
   }

}
