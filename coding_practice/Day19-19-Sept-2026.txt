# Ansible - Deploy using Git 

---

- name: Deploy app from Git for pathnex
  hosts: all
  become: yes

  tasks:
	- name: Install git 
	  yum: 
	     name: git
	     state: present

	- name: Clone repository 
	  git:
	    repo: "https://github.com/pathnex/app.git"
	    dest: "/opt/pathnex-app"

# Kubernetes - Deployment Rolling update (Detailed)

apiversion: apps/v1
kind: deployment
metadata:
	name: pathnex-rollout
spec:
    replicas: 3 
    strategy:
	type: RollingUpdate
	rollingUpdate:
	  maxsurge: 2
	  maxunavailable: 1
    selector:	
       matchlabels:
	  app: rollout
    template:
       metadata:
	  labels:
	     app: rollout

      spec:
	containers:
	   - name: app
	     image: nginx



# Shell Script - Check internet connectivity

#!/bin/bash

ping -c 2 google.com &> /dev/null

if [ $? -eq 0 ]; then
   echo "Internet is available"
else
   echo "No internet connection"
fi



# Jenkins pipeline - manual approval

pipeline {
     agent any
     environment {
	INSTITUTE_NAME = "Pathnex"
	}
     stages {
        stage('Deploy') {
	   steps {
		input message: "Approval deployment for $INSTITUTE_NAME?"
		sh 'echo "Deployment approved and executed!"'	
	}
    }	
}
}
