# Ansible - set docker container for multiple apps

- name: Setup docker container for multiple apps
  host: all
  become: yes

  tasks:
	- name: pull web image
	  docker_image:
	     name: nginx
	     source: pull
	- name: pull database image
	  Docker_image:
	     name: database
	     source: pull
	- name: Run docker web app
	  docker_container:
	      name: nginx
	      image: nginx
	      state: started
	      published_ports:
	 	 - "8080:80"
	- name: Run database app container
	  docker_container:
		name: database-app
		image: database
		state: started
		published_ports:
		  - "3306:3306"

# Kubernets - Deploying statefulset

apiversion: apps/v1
kind: statefulset
metadata:
	name: pathnex-db-statefulset
spec:
  serviceName = "pathnex-db"
  replicas = 2
  selector:
    matchlabels:
	app: pathnex-db
  template:
    metadata:
	labels:
	   app: pathnex-db
    spec:
	containers:
	    - name: db
	      image: mysql:5.7
	      VolumeMounts:
		- mountPath: /var/lib/mysql
		  name: mysql-storage
   volumeClaimTemplates:
      - metadata:
	   - mysql-storage
	spec:
	  accessMode: ["ReadWriteOnce"]
	  resource:
	    requests:
		storage: 1Gi


# Jenkins pipeline - Build and deploy to kubernetes

pipeline {
	agent any
	Envrionment {
	  IMAGE_NAME = "pathnex-web-app"	
	}
	stages{
	   stage('Build Docker image') {
		steps {
		   script {
		      docker.build("${env.IMAGE_NAME}")
			}
		}
	   }
	   stage('Push Docker image') {
		steps {
		   script {
			docker.writeRegistry('http://docker.io', 'docker-credentials') {
			docker.image("${env.IMAGE_NAME}").push('latest')
				}	
			}
		}
	   }
	   stage('deploy to kubernetes') {
		steps {
		   script {
			sh 'kubectl apply -f deployment.yaml'
			}
		   }
	    }

      }

}



# GitLab Ci/CD - build, push, deploy

stages:
  - build
  - push 
  - deploy

build:
	stage: build
	script:
	   - docker build -t pathnex-web-app .

push:
	stage: push
	script:
	   - docker login -u "$CI_REGISTRY_USER" - p "$CI_REGISTRY_PASSWORD"
	   - docker push pathnex-web-app

deploy:
	stage: deploy
	script:
	    - kubectl apply -f kubernetes/deployment.yaml
