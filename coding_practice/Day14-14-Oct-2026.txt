# Terraform - create ec2 instance

resource "aws_vpc" "pathnexvpc" {

	Name = "Pathnex-VPC"

resource "aws_subnet" "pathnexSubnet" {
	vpc_id = aws_vpc.pathnexvpc.id
	cidr_block = "10.10.1.0/24"
	availability_zone = "us-east-1a"

}


resource "aws_route_table" "PathnexRT" {
	name = "PathnexRT"

	route {
	   cidr_block = "0.0.0.0/0"
	   gateway_id = aws_internet_gateway.pathnexIGW
	}
}


resource "aws_internet_gateway" "pathnexIGW" {
	name = "Pathnex-IGW"

# Ansible

- name: install nginx and deploy html file
  hosts: all
  become: yes
 
  tasks:
	- name: install nginx
	  yum:
	     name: nginx
	     state: present
 	- name: start nginx service
	  service:
	     name: nginx
	     state: started
	     enabled: yes
	- name: deploy html file
	  copy:
	    src: files/index.html
	    dest: /var/www/html/index.html
	    owner: root
	    group: root
	    mode: '0644'


# Kubernetes - deployment + service

apiversion: apps/v1
kind: Deployment
metadata:
	name: pathnex-deploy
spec:
  replicas: 2
  selector:
    matchlabels:
	app: pathnex-web
  template:
    metadata:
	labels:
	   app: pathnex-web
    spec:
	containers:
	    - name: web
	      image: nginx:latest
	      ports:
	     	- containerPort: 80
---

apiversion: v1
kind: Service
metadata:
	name: pathnex-nodeport
spec:
   type: Nodeport
   selector:	
	app: pathnex-web
   ports:
    - protocol: TCP
      port: 80
      targetPort: 80
      nodePort: 30247
