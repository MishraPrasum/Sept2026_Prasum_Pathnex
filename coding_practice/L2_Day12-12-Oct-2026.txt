# Oct-L2-Day12:  Scaling application and automation

# Ansible - Setup Autoscaling group

- name: setup autoscaling group
  hosts: all
  become: yes

  tasks:
	- name: Crate launch configuration
	  ec2_asg:
	    name: max-asg
	    region: ap-south-1
	    launch_config_name: max-launch-config
	    min_size: 1
	    max_size: 3
	    desired_capacity: 2
	    vpc_zone_identifier: subnet-0123456789abcdef0
	    

# Terraform - LoadBalancer with EC2 instance

resource "aws_lb" "max_lb" {
	name	= "max_lb"
	internal = false
	load_balancer_type = "application"
	security_groups	= [aws_security_group.allow_ssh_http.id]

# Kubernetes - HPA with Load balancer

apiversion: apps/v1
kind: Deployment
metadata:
	name: pathnec-web
spec:
  replicas: 2
  selector:
    matchlabels:
	app: pathnex-web
  template:
    metadata:
	labels:	
	   app: pathnex-web
    spec:
	containers:	
	   - name: nginx
	     image: nginx
	     ports:
	       - containerPort: 80
---
apiversion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
	name: pathnex-hpa
spec:
   scaleTargetRef:
      apiversion: apps/v1
      kind: Deployment
      name: pathnex-web
   minReplicas: 1
   maxreplicas: 5
   targetCPUUtilizationPercentage: 80


# jenkins pipeline - implement parallel stages

pipeline {
	agent any
	stages{
	   stage('Build & Test' ){
		parallel {
		    stage('Build') {
			steps {	
			  sh 'mvn clean compile'
			     }
		    }
		    stage('Test') {
			steps {
			   sh 'mvn run test'
			      }
		    }
		 }
	   }
	  stage('Deploy') {
		steps {
		   echo "Deploying to staging"
		     }
	 }
     }

}
