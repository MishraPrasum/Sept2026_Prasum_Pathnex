	- name: print OS info

# Kubernetes - Node Affinity 

apiversion: v1
kind: pod
metadata:
	name: pathnex-pod
spec:
  affinity:
   nodeAffinity:
     requiredDuringSchedulingIgnoredDuringExecution:
  	nodeSelectorTerms:
	   matchExpressions:
	      - key: disktype
	        operator: in
	 	values:
	    	  - ssd
  containers:
     - name: web
       image: nginx



# Shell Script: Archive log

#!/bin/bash

tar -czf /backup/pathnex-log-$(date +%F).tar.gz /var/log

echo "Logs archived"


# Jenkins pipeline

pipeline {
	agent any
	environment {
	   INSTITUTE_NAME = "Pathnex"
	}
	stages {
	   stage('Deploy') {
	     steps {
		script {
		    try {
			sh 'echo "Deploying $INSTITUTE_NAME application"'
			sh 'exit 1'
		    } catch (Exception e) {
			echo "Deployment Failed. Rolling back $INSTITUTE_NAME application"
			}
		}
	    }
	 }
     }
}



# GitLab - Rollback

stages:
   - deploy

variables:
   INSTITUTE_NAME = "pathnex"
deploy:
	stage: deploy
	image: alpine:latest
	script:
	   - echo "Deploying $INSTITUTE_NAME application"
	   - exit 1
        after_script:
	   - echo "Deployment failed. Rolling back $INSTITUTE_NAME application"
