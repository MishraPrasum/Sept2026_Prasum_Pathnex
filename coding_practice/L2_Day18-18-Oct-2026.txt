# L2-Day18 - High Availability, Multi-Region, and Auto-scaling

# Ansible 

- name: Install & Configure HAProxy for load balancing 
  hosts: all
  become: yes

  tasks:
      - name: Install HAProxy
        yum:
	  name: HAProxy
	  state: present
      - name: configure HAProxy
	template:
	    src: haproxy.cfg.j2
	    dest: /etc/haproxy/haproxy.cfg
      - name: Start HAProxy service
	service:
	   name: HAProxy
	   state: started
	   enabled: yes


# Terraform — Multi-Region Auto-scaling Group with ALB

resource "aws_lb" "PathnecLB" {
	name	= "Pathnex-LB"
	internal = false
	load_balance_type = "Application"
	security_group = [aws_security_group.allow_ssh_http.id]
	subnets        = [aws_subnet.public.id, aws_subnet.public2.id]
}


resource "aws_autoscaling_group" "PathnexASG" {
	desired_capacity = 3
	max_size	 = 5
	min_size	 = 2
	vpc_zone_identifier = [aws_subnet_public.id, aws_subnet_public2.id]
	launch_configuration = aws_launch_configuration.pathnex_config.id
	
}


# K8s - HPA 

apiversion: apps/v1
kind: Deployment
metadata:
	name: pathnex-hpa
spec:
   replicas: 3
   selector:
     matchlabels:
	app: pathnex-hpa
   template:
	metadata:
	   labels:
	     app: pathnex-hpa
     spec:
	containers:
	    - name: web
	      image: nginx
	      ports:
		- containerPort: 80

---
apiversion: autoscaling/v2
kind: HorizontalPodAutoScaler
metadata:
	name: pathnex-hpa
spec:
   scaleTargetRef:
	apiversion: apps/v1
	kind: Deployment
	name: pathnex-hpa
   minReplicas: 2
   maxReplicas: 10
   targetCPUUtilizationPercentage: 80


#Jenkinsfile — Blue-Green Deployment Strategy

pipeline {
	agent any
	stages{
	   stage('Build') {
	      steps {
		 echo "Building docket image"
		 docker.build('Pathnex-web-app')	
		}
	   }
	   stage('Test') {
		steps {
		   echo  "Running Test"
		 }
	    }
	   stage('Deploy') {
		steps {
		   script {
			sh 'kubectl apply -f kubernetes/blue-deployment.yaml'
			sh 'kubectl apply -f kubernetes/green-deployment.yaml'
			sh 'kubectl rollout status deployment/blue-deployment'
			sh 'kubectl rollout status deployment/green-deployment'
			}
		 }
	  }
     }
}




#GitLab CI/CD — Blue-Green Deployment with Kubernetes

stages:
  - build
  - push
  - deploy

build:
	stage: build
	script:
	   - docker build -t pathnex-web-app .

push:
	stage: push
	script:
	   - docker login -u ""$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
	   - docker push pathnex-web-app

deploy:
	stage: deploy
	script:
	   - kubectl apply -f kubernetes/blue-deployment.yaml
	   - kubectl apply -f kubernetes/green-deployment.yaml
	   - kubectl rollout status deployment/blue-deployment
	   - kubectl rollout status deployment/green-deployment
